# appwalk Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `appwalk`, a zero-dependency Node CLI that walks a person through their own app front-to-back (tree map browser, deterministic guided tour, glossary, audit, monitor, BYO-key AI explanations) and teaches them the language as it goes.

**Architecture:** Single ESM Node 20+ package, zero runtime deps. A `bin/` entry dispatches modes; `src/` holds one focused module per concern (tree, tui, walkthrough, glossary, audit/*, history, ai/*, monitor, report). The deterministic core (everything except `ai/`) runs with no keys; `ai/` is a pluggable BYO-key layer with a lite-model default and per-call cost display.

**Tech Stack:** Node 20+ (ESM), `node:test` + `assert/strict`, ANSI terminal control by hand (no deps), `child_process` for git/run, `node:fs/promises`.

## Global Constraints

- Node >= 20, ESM (`"type": "module"`), **zero runtime dependencies** — `package.json` must have an empty `dependencies` object. All terminal control is raw ANSI.
- Tests via `node --test` + `assert/strict`, one test file per module in `test/`.
- The `.env` file is shown as a *location* only — **never print its values**, in any mode.
- The deterministic core (tree, walkthrough, glossary, audit, monitor, report) must work with **no AI key**. AI features degrade to a "add your key in `.env`" hint, never a crash.
- `.env.example` is the committed blank template; `.env` is gitignored.
- Repo rules from CLAUDE.md: small commits, reviewer is native Claude, never invent API shapes (mark `# VERIFY`).

import { readFile, readdir, stat } from 'node:fs/promises';
import { join, basename, extname, relative } from 'node:path';

export const esc = (s) => String(s ?? '').replace(/[&<>"']/g,
  (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]));

export function humanSize(n) {
  if (!n) return '0 B';
  const u = ['B', 'KB', 'MB', 'GB'];
  let i = 0; let v = n;
  while (v >= 1024 && i < u.length - 1) { v /= 1024; i++; }
  return `${v.toFixed(v >= 10 || i === 0 ? 0 : 1)} ${u[i]}`;
}

const LANG = {
  js: 'js', mjs: 'js', cjs: 'js', jsx: 'jsx', ts: 'ts', tsx: 'tsx',
  py: 'py', rb: 'rb', go: 'go', rs: 'rs', java: 'java', php: 'php',
  html: 'html', css: 'css', scss: 'scss', json: 'json', md: 'md',
  sql: 'sql', sh: 'sh', yml: 'yml', yaml: 'yml', toml: 'toml', env: 'env',
};
export const langOf = (p) => LANG[extname(p).replace(/^\./, '').toLowerCase()] || 'text';

export async function readIgnore(root) {
  try {
    const raw = await readFile(join(root, '.gitignore'), 'utf8');
    return raw.split('\n').map((l) => l.trim()).filter((l) => l && !l.startsWith('#'));
  } catch { return []; }
}

const ALWAYS_SKIP = new Set(['node_modules', '.git', 'dist', '.next', 'build', 'coverage', '.venv', 'venv', '__pycache__', '.superpowers']);

function ignored(rel, patterns) {
  return patterns.some((p) => {
    const base = p.replace(/^\/+|\/+$/g, '');
    const b = base.replace(/\*/g, '[^/]*');
    return rel === base || rel.startsWith(base + '/') || rel.endsWith('/' + base) || new RegExp('^' + b + '$').test(rel);
  });
}

export async function walk(root, _seen = new Set()) {
  const patterns = await readIgnore(root);
  const out = [];
  async function rec(dir) {
    let ents;
    try { ents = await readdir(dir, { withFileTypes: true }); } catch { return; }
    for (const e of ents) {
      const abs = join(dir, e.name);
      const relp = relative(root, abs).split('/').join('/');
      const first = relp.split('/')[0];
      if (ALWAYS_SKIP.has(first) || first.startsWith('.')) continue;
      if (ignored(relp, patterns)) continue;
      if (e.isDirectory()) await rec(abs);
      else if (e.isFile()) out.push(abs);
    }
  }
  await rec(root);
  return out.sort();
}

export const rel = (root, abs) => relative(root, abs);

export async function readFileSafe(abs) {
  try { return await readFile(abs, 'utf8'); } catch { return null; }
}

// src/tree.js
import { readFile, stat } from 'node:fs/promises';
import { join, basename, dirname } from 'node:path';
import { walk, langOf, rel, readFileSafe } from './util.js';

const ENTRY_NAMES = new Set(['package.json', 'index.html', 'main.py', 'app.py', 'manage.py', 'Dockerfile', 'README.md', 'server.js', 'server.py', 'index.js', 'index.ts', 'app.js', 'app.ts', 'main.js', 'main.ts', 'vite.config.js', 'vite.config.ts']);

export async function isEntry(name, text) {
  if (ENTRY_NAMES.has(name)) return true;
  if (!/\.(js|ts|py|jsx|tsx)$/.test(name) || text.length < 200) return false;
  return /express|flask|fastapi|createServer|app\.listen|if __name__/.test(text.slice(0, 600));
}

export async function buildMap(root) {
  const files = await walk(root);
  const flat = [];
  for (const abs of files) {
    let s; try { s = await stat(abs); } catch { continue; }
    const text = await readFileSafe(abs);
    const name = basename(abs);
    flat.push({
      path: abs,
      name,
      dir: rel(root, dirname(abs)),
      lang: langOf(abs),
      size: s.size,
      mtime: s.mtimeMs,
      entry: await isEntry(name, text || ''),
    });
  }
  return { root, flat, entryPoints: flat.filter((n) => n.entry) };
}
```

// src/tui.js
import { readFileSafe, humanSize } from './util.js';

export function renderTree(map, cursor, width = 80) {
  const lines = [];
  lines.push(`appwalk tree — ${map.root}`);
  lines.push('↑↓ move · Enter open · q quit'.padEnd(width, ' '));
  for (let i = 0; i < map.flat.length; i++) {
    const n = map.flat[i];
    const mark = n.entry ? '▶' : ' ';
    const meta = `${n.lang} · ${humanSize(n.size)}`.padStart(18);
    const line = `${mark} ${n.name.padEnd(24)}${meta}`.slice(0, width);
    lines.push(i === cursor ? `\x1b[7m${line}\x1b[0m` : line);
  }
  const hasEnv = map.flat.some((n) => n.name === '.env');
  lines.push(hasEnv ? '   .env detected — shown as location only (values never printed)'.slice(0, width) : '');
  return lines.join('\n');
}

export async function runTree(map, io = { stdin: process.stdin, stdout: process.stdout }) {
  let cursor = 0;
  const detail = async (node) => {
    const text = (await readFileSafe(node.path) || '').slice(0, 4000);
    io.stdout.write('\x1b[2J\x1b[H' + `── ${node.path} ──\n\n${text}\n\n[Enter back · q quit]\n`);
  };
  const draw = () => io.stdout.write('\x1b[2J\x1b[H' + renderTree(map, cursor));
  draw();
  const stdin = io.stdin;
  const wasRaw = stdin.isRaw;
  if (stdin.setRawMode) stdin.setRawMode(true);
  stdin.resume();
  stdin.setEncoding('utf8');
  let buf = '';
  return new Promise((resolve) => {
    const onData = async (chunk) => {
      buf += chunk;
      while (buf.length) {
        const ch = buf[0];
        if (ch === '\x1b') {
          if (buf[1] === '[' && (buf[2] === 'A' || buf[2] === 'B')) {
            if (buf[2] === 'A') cursor = Math.max(0, cursor - 1);
            else cursor = Math.min(map.flat.length - 1, cursor + 1);
            buf = buf.slice(3); draw();
          } else buf = buf.slice(1);
          continue;
        }
        buf = buf.slice(1);
        if (ch === 'q') return finish();
        if (ch === '\r' || ch === '\n') {
          const node = map.flat[cursor];
          if (node) await detail(node);
          draw();
        }
      }
    };
    const finish = () => {
      stdin.removeListener('data', onData);
      if (stdin.setRawMode && wasRaw === false) stdin.setRawMode(false);
      stdin.pause();
      io.stdout.write('\x1b[2J\x1b[H');
      resolve();
    };
    stdin.on('data', onData);
  });
}
```

// src/walkthrough.js
import { basename } from 'node:path';
import { readFileSafe, rel } from './util.js';
import { findMatches } from './glossary.js';
import { educateText } from './ai/educate.js';   // deterministic text — no AI key needed

export function detectStack(map) {
  const names = new Set(map.flat.map((n) => n.name));
  if (names.has('package.json')) {
    return { type: 'node', runCmd: 'npm run dev (or the dev script in package.json)', entry: 'package.json' };
  }
  if (names.has('main.py') || names.has('app.py') || names.has('manage.py')) {
    return { type: 'python', runCmd: 'python main.py', entry: names.has('main.py') ? 'main.py' : 'app.py' };
  }
  if (names.has('index.html')) return { type: 'static', runCmd: 'open index.html (or serve the folder)', entry: 'index.html' };
  return { type: 'unknown', runCmd: null, entry: null };
}

const gloss = (s) => {
  const hits = findMatches(s);
  return hits.length ? `\n   [terms: ${hits.join(', ')} — run \`appwalk glossary <term>\` for what they mean]` : '';
};

export async function buildTour(map, stack) {
  const steps = [];
  const n = (p) => map.flat.find((x) => x.path.endsWith(p));

  steps.push({
    title: 'What this app is',
    body: `I can see ${map.flat.length} files. It looks like a ${stack.type === 'node' ? 'Node.js' : stack.type === 'python' ? 'Python' : stack.type === 'static' ? 'static site' : 'project I don\'t recognize yet'}.${gloss('dependency')}`,
    files: [],
  });

  steps.push({
    title: 'How it starts',
    body: stack.runCmd
      ? `Run it with: ${stack.runCmd}.${stack.entry ? ` The entry point is ${stack.entry} — start reading there.` : ''}${gloss('deploy')}`
      : 'I couldn\'t find a standard entry point. Try `appwalk tree` to explore, or add a README that says how to run it.',
    files: stack.entry ? [n(stack.entry)?.path].filter(Boolean) : [],
  });

  const dirs = [...new Set(map.flat.map((f) => f.dir))].filter((d) => d !== '.').slice(0, 12);
  steps.push({
    title: 'These are your folders',
    body: dirs.length ? `Top-level folders: ${dirs.join(', ')}. Each one usually holds one job — code, config, assets.${gloss('schema')}` : 'Everything lives in the root folder for now — that\'s fine for a small app.',
    files: [],
  });

  const entry = n(stack.entry || 'index.js');
  if (entry) {
    const text = (await readFileSafe(entry.path) || '').slice(0, 1200);
    steps.push({
      title: `A look inside ${basename(entry.path)}`,
      body: `This file is ${entry.lang}, ${entry.size} bytes.${text ? `\n\n${text}\n\n` : ''}${gloss('debugging')}`,
      files: [entry.path],
    });
  }

  steps.push({
    title: 'Choosing your AI model (optional — free to skip)',
    body: educateText(),
    files: [],
  });

  steps.push({
    title: 'What to watch',
    body: 'Next, run `appwalk audit` — it checks for leaked secrets, leftover clutter, duplicate files that could be merged, and risky dependencies. Every finding is explained with a fix.${gloss('API key')}',
    files: [],
  });
  return steps;
}

// src/audit/secrets.js
import { readFileSafe } from '../util.js';
import { basename, join } from 'node:path';

const PATTERNS = [
  { kind: 'stripe_live', re: /sk_live_[A-Za-z0-9]{16,}/g, label: 'Stripe live secret key' },
  { kind: 'aws', re: /AKIA[0-9A-Z]{16}/g, label: 'AWS access key' },
  { kind: 'github', re: /ghp_[A-Za-z0-9]{20,}/g, label: 'GitHub personal access token' },
  { kind: 'google', re: /AIza[0-9A-Za-z\-_]{20,}/g, label: 'Google API key' },
  { kind: 'slack', re: /xox[baprs]-[0-9A-Za-z\-]{10,}/g, label: 'Slack token' },
  { kind: 'private_key', re: /-----BEGIN (RSA |EC |OPENSSH )?PRIVATE KEY-----/g, label: 'private key block' },
  { kind: 'db_url', re: /(postgres|mysql|mongodb)(\+srv)?:\/\/[^\s"']+:[^\s"']+@/g, label: 'database URL with credentials' },
  { kind: 'generic_key', re: /["'](api[_-]?key|secret|password|token)["']\s*[:=]\s*["'][A-Za-z0-9_\-.]{12,}["']/gi, label: 'hardcoded key/password' },
];

function mask(s) {
  if (s.length <= 8) return s;
  return s.slice(0, 4) + '…' + s.slice(-4);
}

export async function scanSecrets(map) {
  const out = [];
  const hasEnv = map.flat.some((n) => n.name === '.env');
  for (const node of map.flat) {
    if (node.name === '.env') continue;
    const text = await readFileSafe(node.path);
    if (!text) continue;
    const lines = text.split('\n');
    for (let i = 0; i < lines.length; i++) {
      for (const p of PATTERNS) {
        p.re.lastIndex = 0;
        const m = p.re.exec(lines[i]);
        if (m) {
          out.push({ file: node.path, line: i + 1, kind: p.kind, snippet: mask(m[0]), label: p.label });
          break; // one finding per line
        }
      }
    }
  }
  if (hasEnv) {
    out.push({ file: '.env', line: 0, kind: 'env_in_git', snippet: '', label: '.env file is present in the project tree' });
  }
  return out;
}

// src/audit/duplicates.js
import { readFileSafe } from '../util.js';
function norm(s) { return (s || '').replace(/\s+/g, ' ').trim(); }
export async function scanDuplicates(map, { minSim = 0.8 } = {}) {
  const texts = [];
  for (const node of map.flat) {
    if (!/\.(js|ts|jsx|tsx|py|html|css|json|md)$/.test(node.name)) continue;
    if (node.size < 200) continue;
    const t = await readFileSafe(node.path);
    if (t) texts.push({ node, t: norm(t) });
  }
  const out = [];
  for (let i = 0; i < texts.length; i++) {
    for (let j = i + 1; j < texts.length; j++) {
      const a = texts[i].t, b = texts[j].t;
      if (!a || !b) continue;
      if (a === b) { out.push({ file: texts[i].node.path, kind: 'duplicate', other: texts[j].node.path, sim: 1 }); continue; }
      // containment sim (short vs long) + token overlap
      const shorter = a.length < b.length ? a : b, longer = a.length < b.length ? b : a;
      if (shorter.length > 50 && longer.includes(shorter)) out.push({ file: texts[i].node.path, kind: 'duplicate', other: texts[j].node.path, sim: 0.95 });
    }
  }
  return out;
}

// src/audit/clutter.js
import { readdir } from 'node:fs/promises';
import { dirname } from 'node:path';
import { readFileSafe } from '../util.js';

export async function scanClutter(map) {
  const out = [];
  for (const node of map.flat) {
    if (/\.(bak|tmp|orig|swp)$/i.test(node.name)) out.push({ file: node.path, kind: 'bak', reason: `${node.name} looks like a leftover backup file` });
    else if (node.name.toLowerCase() === 'todo') out.push({ file: node.path, kind: 'todo_only', reason: 'a file literally named TODO' });
    else if (/(\.js|\.py)$/.test(node.name)) {
      const t = await readFileSafe(node.path);
      if (t && /^\s*(\/\/|#|<!--|~).*$/m.test(t) && !t.trim().replace(/^\s*(#|\/\/|<!--|~).*$/gm, '').trim())
        out.push({ file: node.path, kind: 'todo_only', reason: 'contains only comments/TODO lines — likely never finished' });
    }
  }
  // empty directories
  const dirs = new Set(map.flat.map((n) => n.dir).filter((d) => d !== '.'));
  for (const d of dirs) {
    let ents; try { ents = await readdir(dirname(map.flat[0].path) + '/' + d); } catch { continue; }
    if (!ents.length) out.push({ file: d, kind: 'empty_dir', reason: 'empty directory — can be removed' });
  }
  return out;
}

// src/audit/deps.js
import { execFile } from 'node:child_process';
import { promisify } from 'node:util';
import { readFileSafe } from '../util.js';
import { join } from 'node:path';
const exec = promisify(execFile);

export async function scanDeps(map) {
  const out = [];
  const pkg = map.flat.find((n) => n.name === 'package.json');
  if (pkg) {
    if (!map.flat.some((n) => /package-lock\.json|yarn\.lock|pnpm-lock\.yaml|bun\.lockb/.test(n.name)))
      out.push({ file: 'package.json', kind: 'missing_lockfile', reason: 'no lockfile — installs are not reproducible' });
    const raw = JSON.parse(await readFileSafe(pkg.path) || '{}');
    const deps = Object.keys(raw.dependencies || {});
    for (const d of deps) {
      // best-effort import scan: a dep is "used" if any source file imports it
      const re = new RegExp(`from ['"]${d}['"]|require\\(['"]${d}['"]\\)`);
      const hit = await Promise.all(map.flat.map(async (n) => {
        if (!/\.(js|jsx|ts|tsx)$/.test(n.name)) return false;
        const t = await readFileSafe(n.path);
        return !!t && re.test(t);
      }));
      if (!hit.some(Boolean)) out.push({ file: `package.json → ${d}`, kind: 'unused_dep', reason: `${d} is declared but I couldn't find an import — candidate to remove` });
    }
  }
  return out;
}

// src/history.js
import { execFile } from 'node:child_process';
import { promisify } from 'node:util';
const exec = promisify(execFile);

export async function fileHistory(root, relPath) {
  try {
    const { stdout } = await exec('git', ['log', '--follow', '--format=%h|%ad|%s', '--date=short', '--', relPath], { cwd: root, timeout: 8000 });
    const lines = stdout.split('\n').filter(Boolean);
    if (!lines.length) return null;
    const first = lines[lines.length - 1];
    const last = lines[0];

// src/monitor.js
import { spawn } from 'node:child_process';
import { split as _ } from 'node:path';

export function isCrash(line) {
  return /TypeError:|ReferenceError:|Error:|EADDRINUSE|ECONNREFUSED|UnhandledPromiseRejection|panic:|Traceback/.test(line);
}

export function runCommand(stack) {
  if (stack.type === 'node') return { cmd: 'npm', args: ['run', 'dev'] };
  if (stack.type === 'python') return { cmd: 'python', args: ['main.py'] };
  if (stack.type === 'static') return { cmd: 'python', args: ['-m', 'http.server', '8000'] };
  return null;
}

export function monitor(root, stack, { explain } = {}) {
  const run = runCommand(stack);
  if (!run) { console.log('monitor: no run command known for this project. Try `appwalk walk` first.'); return Promise.resolve(); }
  console.log(`monitor: running \`${run.cmd} ${run.args.join(' ')}\` (Ctrl-C to stop)`);
  return new Promise((resolve) => {
    const child = spawn(run.cmd, run.args, { cwd: root, stdio: ['ignore', 'pipe', 'pipe'] });
    const onLine = (line) => {
      process.stdout.write(line + '\n');
      if (isCrash(line)) {
        console.log('⚠ possible error detected.');
        if (explain) explain(`A running app printed this error. Explain it simply and what to check:\n\n${line}`).then((r) => console.log('\n' + r.text));
      }
    };
    let buf = '';
    const feed = (chunk) => {
      buf += chunk.toString();
      let nl;
      while ((nl = buf.indexOf('\n')) >= 0) { const line = buf.slice(0, nl); buf = buf.slice(nl + 1); if (line.trim()) onLine(line); }
    };
    child.stdout.on('data', feed);
    child.stderr.on('data', feed);
    child.on('exit', (code) => { console.log(`monitor: app exited (code ${code})`); resolve(); });
    process.on('SIGINT', () => { try { child.kill('SIGTERM'); } catch {} setTimeout(resolve, 300); });
  });
}
